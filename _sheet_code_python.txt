 | def est_premier(n):
 |     if n<2: return False
 |     if n==2: return True
 |     if n%2==0: return False
 |     for i in range(3,int(n**0.5)+1,2):
 |         if n%i==0: return False
 |     return True

 | def candidats_premiers_geo(k):
 |     tA=suite_geo_A(k); sA=sum(tA)
 |     tB=suite_geo_B(k); sB=sum(tB)
 |     cands=set()
 |     for v in [sA,sB,sA+sB,abs(sA-sB)]:
 |         for c in [math.floor(v),math.ceil(v),round(v)]:
 |             if c>1: cands.add(c)
 |     for n in range(1,15):
 |         for v in [n*sA,n*sB]:
 |             c=round(v)
 |             if 2<=c<=10_000_000: cands.add(c)
 |     return sorted(p for p in cands if est_premier(p))

 | def reconstruire_avec_fallback_geo(n, fn_algebrique, k=2):
