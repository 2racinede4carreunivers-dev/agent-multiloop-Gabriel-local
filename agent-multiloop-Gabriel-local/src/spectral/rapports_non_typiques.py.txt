Moteur de reconstruction des nombres premiers pour les rapports non-typiques

  NIVEAU 1 — ENTIER (k^n)

def _is_prime(p: int) -> bool:
    """Test de primalité entière exact (sans table bornée)."""
    if not isinstance(p, int) or p < 2:
        return False
    if p == 2:
        return True
    if p % 2 == 0:
        return False
    return all(p % d != 0 for d in range(3, isqrt(p) + 1, 2))


def _prime_table(count: int) -> List[int]:
    """Génère la table des `count` premiers nombres premiers."""
    out: List[int] = []
    p = 2
    while len(out) < count:
        if _is_prime(p):
            out.append(p)
        p = 3 if p == 2 else p + 2
    return out


def _extraire_k(rapport: Union[str, int]) -> int:
    """Extrait k depuis '1/k' (str) ou un entier direct."""
    if isinstance(rapport, int):
        return rapport
    s = str(rapport).strip()
    if s.startswith("1/"):
        return int(s[2:])
    try:
        return int(s)
    except ValueError:
        raise ValueError(f"Format de rapport invalide : '{rapport}' (attendu '1/k' ou entier k)")


# ══════════════════════════════════════════════════════════════════════════════
# SECTION 3 — NIVEAU ENTIER : CONSTRUCTION DES SUITES A ET B (k^n)
# ══════════════════════════════════════════════════════════════════════════════
# Règles de Savard (methode_spectral.thy § XI + § XII) :
#   Suite A : k^i pour i≤n-2 ; T(n-1)=k^(n-1)-k^(n-3) ; T(n)=k^n-k^(n-2)
#   Suite B : k^i pour i≤5 ; SAUT ZÊTA à i=6→k^7 ; k^(i+1) pour i≥7 ;
#             T(n-1)=k^n-k^(n-2) ; T(n)=k^(n+1)-k^(n-1)

def term_a(k: int, n: int, i: int) -> int:
    """Terme i de la suite A pour le rapport 1/k avec n termes."""
    if not (k >= 2 and n >= 1 and 1 <= i <= n):
        return 0
    if n <= 7:
        return k ** i
    if i <= n - 2:
        return k ** i
    if i == n - 1:
        return k ** (n - 1) - k ** (n - 3)
    return k ** n - k ** (n - 2)          # i == n


def term_b(k: int, n: int, i: int) -> int:
    """Terme i de la suite B pour le rapport 1/k avec n termes.

    Le Saut Zêta (position 6 → k^7 au lieu de k^6) est la connexion
    qualitative au comportement de ζ documentée en § XI de methode_spectral.thy.
    """
    if not (k >= 2 and n >= 1 and 1 <= i <= n):
        return 0
    if n <= 7:
        return term_a(k, n, i)
    if i <= 5:
        return k ** i
    if i == 6:
        return k ** 7                      # ← SAUT ZÊTA
    if i <= n - 2:
        return k ** (i + 1)
    if i == n - 1:
        return k ** n - k ** (n - 2)
    return k ** (n + 1) - k ** (n - 1)    # i == n


def build(k: int, n: int) -> Tuple[List[int], List[int]]:
    """Construit les listes complètes A et B pour le rapport 1/k à n termes."""
    A = [term_a(k, n, i) for i in range(1, n + 1)]
    B = [term_b(k, n, i) for i in range(1, n + 1)]
    return A, B


def closed_sums(k: int, n: int) -> Tuple[Fraction, Fraction]:
    """Formules fermées exactes (Fraction) pour les sommes SA et SB.

    αA(k) = (k⁴-k²+1) / [(k-1)·k³]
    SA(k,n) = αA(k)·k^n − k/(k−1)
    SB(k,n) = k·αA(k)·k^n − (k⁷-k⁶+k)/(k−1)

    Source : onglet "Modèle Formel" + "Systeme General" du fichier Excel v7.4.
    """
    h = Fraction(k ** 4 - k ** 2 + 1, (k - 1) * k ** 3)
    sa = h * k ** n - Fraction(k, k - 1)
    sb = k * h * k ** n - Fraction(k ** 7 - k ** 6 + k, k - 1)
    return sa, sb


# ══════════════════════════════════════════════════════════════════════════════
# SECTION 4 — QUATRE ESSAIS DIGAMMA (ancrage n=10)
# ══════════════════════════════════════════════════════════════════════════════
# Règle spéciale Savard : à n=10, on teste les 4 combinaisons
# (position ∈ {7,8}, signe ∈ {+1,-1}) pour le terme Digamma.
# Le premier candidat premier valide devient l'ancrage du rapport.
#
# Preuve par l'absurde HOL : methode_spectral.thy § XIII + Démonstration Formelle

def digamma_trials_n10(k: int) -> List[Dict]:
    """Quatre essais Digamma à n=10 pour le rapport 1/k.

    Retourne une liste de 4 dict avec les champs :
      position, signe, digamma, P, premier (bool)
    """
    A, B = build(k, 10)
    sa, sb = sum(A), sum(B)
    out = []
    for pos in (7, 8):
        for sign in (+1, -1):
            d = Fraction(sa + sign * A[pos - 1])
            q = Fraction(sb - d, k ** 6)
            p = q.numerator if q.denominator == 1 else q
            is_p = isinstance(p, int) and _is_prime(p)
            out.append({
                "position": pos,
                "signe":    sign,
                "digamma":  int(d),
                "P":        int(p) if isinstance(p, int) else p,
                "premier":  is_p,
            })
    return out


def _ancrage_valide(k: int) -> Tuple[int, int]:
    """Retourne (premier, rang) de l'ancrage n=10 pour le rapport 1/k.

    Priorité : ANCHORS → CATALOGUE_DIGAMMA → essais dynamiques → erreur.
    """
    if k in ANCHORS:
        return ANCHORS[k]
    if k in CATALOGUE_DIGAMMA:
        return CATALOGUE_DIGAMMA[k]
    # Tentative dynamique par les quatre essais
    trials = digamma_trials_n10(k)
    premiers_trouves = [t for t in trials if t["premier"]]
    if len(premiers_trouves) == 1:
        p = premiers_trouves[0]["P"]
        # Calculer le rang (coûteux mais nécessaire)
        primes = _prime_table(100_000)
        if p in primes:
            rang = primes.index(p) + 1
            logger.info("Ancrage dynamique k=%d : P=%d rang=%d", k, p, rang)
            return (p, rang)
    raise ValueError(
        f"Rapport 1/{k} : ancrage n=10 introuvable. "
        f"Essais : {[t['P'] for t in trials]} — "
        f"Ajouter manuellement dans ANCHORS ou CATALOGUE_DIGAMMA."
    )


# ══════════════════════════════════════════════════════════════════════════════
# SECTION 5 — RECONSTRUCTION ENTIÈRE PAR RANG (niveau entier)
# ══════════════════════════════════════════════════════════════════════════════

def reconstruct_entier(k: int, n: int,
                       primes: Optional[List[int]] = None) -> Dict:
    """Reconstruit le premier pour le rapport 1/k à n termes (niveau entier).

    Algorithme :
      1. Ancrage (premier_ancrage, rang_ancrage) à n=10
      2. rang_cible = rang_ancrage + (n - 10)
      3. p = primes[rang_cible - 1]
      4. Digamma = SB(k,n) - p·k⁶  (calcul à rebours)
      5. Assertion : (SB - Digamma) / k⁶ == p  (vérification exacte)

    Retourne un dict complet avec toutes les données convolutives.
    """
    base_p, base_rank = _ancrage_valide(k)
    target_rank = base_rank + (n - 10)
    if target_rank < 1:
        raise ValueError(f"Rang cible {target_rank} < 1 pour k={k}, n={n}")

    if primes is None or len(primes) < target_rank:
        primes = _prime_table(target_rank)

    p = primes[target_rank - 1]
    A, B = build(k, n)
    sa = sum(A)
    sb = sum(B)
    d = sb - p * (k ** 6)

    # Vérification exacte (Fraction pour éviter tout arrondi)
    check = Fraction(sb - d, k ** 6)
    assert check == p, (
        f"Assertion reconstruction échouée k={k} n={n} : "
        f"(SB-D)/k⁶ = {check} ≠ {p}"
    )

    sa_closed, sb_closed = closed_sums(k, n)

    return {
        "rapport":       f"1/{k}",
        "k":             k,
        "n":             n,
        "niveau":        "entier",
        "A":             A,
        "B":             B,
        "somme_A":       sa,
        "somme_B":       sb,
        "somme_A_fermee": str(sa_closed),
        "somme_B_fermee": str(sb_closed),
        "digamma":       d,
        "premier":       p,
        "rang_premier":  target_rank,
        "base_premier":  base_p,
        "base_rang":     base_rank,
        "k6":            k ** 6,
        "formule_reconstruction": f"P = (SB - Digamma) / k⁶ = ({sb} - {d}) / {k**6} = {p}",
        "premier_indetermine": False,
        "fallback":      False,
    }


# ══════════════════════════════════════════════════════════════════════════════
# SECTION 6 — NIVEAU GÉOMÉTRIQUE (√(a²+b²)) + FALLBACK GABRIEL v7.4
# ══════════════════════════════════════════════════════════════════════════════
# Facteur commun g(k) = √(1+k²)/k
# S_A,géo = g(k) · S_A,entier  (annulation dans la reconstruction → même P)
# Source : onglet "Suites Géo A-B" + "Moteur Géo Convolutif" + "Code Python"

def facteur_g(k: int) -> float:
    """Facteur de mise à l'échelle géométrique g(k) = √(1+k²)/k."""
    return math.sqrt(1 + k * k) / k


def suite_geo_A(k: int) -> List[float]:
    """Suite géométrique A pour le rapport 1/k (10 termes).

    Termes 1-8 : √(t^(i-1)² + t^i²) avec t = 1/k
    Termes 9-10 : différentiels (√((t^8-t^6)²+(t^9-t^7)²), etc.)
    """
    t = 1.0 / k
    p = [1.0] + [t ** n for n in range(1, 13)]
    return [
        math.sqrt(p[0] ** 2 + p[1] ** 2),
        math.sqrt(p[1] ** 2 + p[2] ** 2),
        math.sqrt(p[2] ** 2 + p[3] ** 2),
        math.sqrt(p[3] ** 2 + p[4] ** 2),
        math.sqrt(p[4] ** 2 + p[5] ** 2),
        math.sqrt(p[5] ** 2 + p[6] ** 2),
        math.sqrt(p[6] ** 2 + p[7] ** 2),
        math.sqrt(p[7] ** 2 + p[8] ** 2),
        math.sqrt((p[8] - p[6]) ** 2 + (p[9]  - p[7])  ** 2),   # T9 diff
        math.sqrt((p[9] - p[7]) ** 2 + (p[10] - p[8])  ** 2),   # T10 diff
    ]


def suite_geo_B(k: int) -> List[float]:
    """Suite géométrique B pour le rapport 1/k (10 termes).

    Même structure que A mais avec le SAUT ZÊTA à la position 6
    (terme (t^5,t^6) absent — remplacé par (t^6,t^7)).
    """
    t = 1.0 / k
    p = [1.0] + [t ** n for n in range(1, 13)]
    return [
        math.sqrt(p[0]  ** 2 + p[1]  ** 2),
        math.sqrt(p[1]  ** 2 + p[2]  ** 2),
        math.sqrt(p[2]  ** 2 + p[3]  ** 2),
        math.sqrt(p[3]  ** 2 + p[4]  ** 2),
        math.sqrt(p[4]  ** 2 + p[5]  ** 2),
        # SAUT ZÊTA géométrique : on saute le couple (p[5],p[6]) → (p[6],p[7])
        math.sqrt(p[6]  ** 2 + p[7]  ** 2),
        math.sqrt(p[7]  ** 2 + p[8]  ** 2),
        math.sqrt(p[8]  ** 2 + p[9]  ** 2),
        math.sqrt((p[10] - p[8])  ** 2 + (p[11] - p[9])  ** 2),  # T9 diff
        math.sqrt((p[11] - p[9])  ** 2 + (p[12] - p[10]) ** 2),  # T10 diff
    ]


def candidats_premiers_geo(k: int) -> List[int]:
    """Candidats premiers issus des combinaisons des sommes géométriques.

    Explore : sA, sB, sA+sB, |sA-sB| et multiples entiers 1..14 de chaque.
    Retourne la liste triée des candidats premiers dans [2, 10_000_000].
    """
    tA = suite_geo_A(k)
    tB = suite_geo_B(k)
    sA = sum(tA)
    sB = sum(tB)
    cands: set[int] = set()

    for v in (sA, sB, sA + sB, abs(sA - sB)):
        for c in (math.floor(v), math.ceil(v), round(v)):
            if c > 1:
                cands.add(c)

    for multiplier in range(1, 15):
        for v in (multiplier * sA, multiplier * sB):
            c = round(v)
            if 2 <= c <= 10_000_000:
                cands.add(c)

    return sorted(c for c in cands if _is_prime(c))


def reconstruct_geometrique(k: int, n: int) -> Dict:
    """Reconstruction niveau géométrique pour le rapport 1/k.

    Le facteur g(k) s'annule dans le calcul de P — le premier reconstruit
    est identique au niveau entier quand l'ancrage est disponible.
    Ce niveau sert de fallback indépendant pour les grands k.
    """
    tA = suite_geo_A(k)
    tB = suite_geo_B(k)
    sA_geo = sum(tA)
    sB_geo = sum(tB)
    g = facteur_g(k)

    # Digamma géométrique : même structure, mis à l'échelle par g(k)
    # (SB_geo - Digamma_geo) / (k^6 * g) = P
    candidats = candidats_premiers_geo(k)
    premier_geo = candidats[0] if candidats else None

    # Correspondance niveau entier (vérification du facteur g)
    try:
        sa_int = float(closed_sums(k, n)[0])
        sb_int = float(closed_sums(k, n)[1])
        g_verif = sA_geo / sa_int if abs(sa_int) > 1e-9 else None
    except Exception:
        g_verif = None

    return {
        "rapport":           f"1/{k}",
        "k":                 k,
        "n":                 n,
        "niveau":            "geometrique",
        "suite_geo_A":       [round(x, 10) for x in tA],
        "suite_geo_B":       [round(x, 10) for x in tB],
        "somme_geo_A":       round(sA_geo, 10),
        "somme_geo_B":       round(sB_geo, 10),
        "facteur_g":         round(g, 10),
        "facteur_g_verifie": round(g_verif, 10) if g_verif else None,
        "candidats_premiers":candidats,
        "premier":           premier_geo,
        "premier_indetermine": premier_geo is None,
        "fallback":          True,
        "note": (
            f"Niveau géométrique actif (g(k)=√(1+{k}²)/{k}={g:.8f}). "
            f"Facteur s'annule dans P → même premier que niveau entier."
        ),
    }


def reconstruire_avec_fallback_geo(k: int, n: int,
                                   primes: Optional[List[int]] = None) -> Dict:
    """Reconstruit P pour 1/k à n termes — Ordre Gabriel v7.4 :

    1. Niveau entier  (term_a/term_b + ancrage + 4 Digamma)
    2. Si échec → Niveau géométrique (suite_geo_A/B + candidats)

    Toujours retourne un dict avec les deux niveaux documentés.
    """
    resultat_entier: Optional[Dict] = None
    erreur_entier: Optional[str] = None

    # ── NIVEAU ENTIER ────────────────────────────────────────────────────────
    try:
        resultat_entier = reconstruct_entier(k, n, primes)
        logger.debug("Niveau entier réussi k=%d n=%d P=%d",
                     k, n, resultat_entier["premier"])
    except Exception as exc:
        erreur_entier = str(exc)
        logger.info("Niveau entier impossible k=%d n=%d : %s", k, n, exc)

    if resultat_entier and resultat_entier.get("premier") is not None:
        # Ajouter les métadonnées géo pour documentation
        tA = suite_geo_A(k)
        tB = suite_geo_B(k)
        resultat_entier["facteur_g"]    = round(facteur_g(k), 10)
        resultat_entier["somme_geo_A"]  = round(sum(tA), 10)
        resultat_entier["somme_geo_B"]  = round(sum(tB), 10)
        resultat_entier["approche"]     = "entier"
        resultat_entier["erreur_entier"]= None
        return resultat_entier

    # ── NIVEAU GÉOMÉTRIQUE (FALLBACK) ────────────────────────────────────────
    logger.info("FALLBACK géométrique activé k=%d n=%d", k, n)
    resultat_geo = reconstruct_geometrique(k, n)
    resultat_geo["approche"]      = "geometrique"
    resultat_geo["erreur_entier"] = erreur_entier
    return resultat_geo

# SECTION 7 — INFORMATIONS STRUCTURELLES SUR LES ÉQUATIONS
# ══════════════════════════════════════════════════════════════════════════════

def _info_equations(k: int) -> Dict:
    """Retourne les équations fermées et les paramètres Digamma pour le rapport 1/k."""
    h_num = k ** 4 - k ** 2 + 1
    h_den = (k - 1) * k ** 3
    alpha_A = f"({h_num}/{h_den})"
    alpha_B = f"({k * h_num}/{h_den})"
    offset_A = f"-{k}/{k-1}"
    offset_B = f"-({k**7 - k**6 + k}/{k-1})"

    pos, sgn = DIGAMMA_PARAMS.get(k, (8, +1))
    signe_str = "+" if sgn == +1 else "−"

    return {
        "equation_A": {
            "nom":    "A",
            "forme":  f"SA(k={k}, n) = {alpha_A} · {k}^n {offset_A}",
            "alpha":  alpha_A,
            "offset": offset_A,
        },
        "equation_B": {
            "nom":    "B",
            "forme":  f"SB(k={k}, n) = {alpha_B} · {k}^n {offset_B}",
            "alpha":  alpha_B,
            "offset": offset_B,
        },
        "digamma_params": {
            "position": pos,
            "signe":    sgn,
            "formule":  f"D = SA {signe_str} A[{pos}]",
        },
        "reconstruction": f"P = (SB − Digamma) / {k}^6",
        "facteur_g":      f"√(1+{k}²)/{k} = {round(facteur_g(k), 8)}",
        "saut_zeta":      f"Position 6 de B : k^7 au lieu de k^6 (connexion ζ)",
    }


# ══════════════════════════════════════════════════════════════════════════════
# SECTION 8 — API PUBLIQUE (compatible spectral_core.py)
# ══════════════════════════════════════════════════════════════════════════════

def construire_rapport_convolutif(rapport: Union[str, int],
                                   n: int = 10) -> Dict:
    """Produit les faits convolutifs complets pour une requête 1/k.

    Retourne un dict standardisé avec :
      - equations A et B (formes fermées)
      - référence n=10 (ancrage validé)
      - reconstruction cible à n=n (niveau entier + métadonnées géo)
      - statut primer_indetermine, fallback, approche
      - métadonnées Pont Savard (facteur_g, saut_zeta, digamma_params)

    Compatibilité : remplace construire_rapport_convolutif() de l'ancienne
    version — même nom, même signature, dict enrichi.
    """
    k = _extraire_k(rapport)

    # Référence ancrage n=10
    try:
        ref_n10 = reconstruct_entier(k, 10)
    except Exception as exc:
        ref_n10 = {"erreur": str(exc), "premier": None, "premier_indetermine": True}

    # Reconstruction cible
    res = reconstruire_avec_fallback_geo(k, n)

    # Essais Digamma à n=10 (documentation)
    try:
        essais = digamma_trials_n10(k)
    except Exception:
        essais = []

    return {
        "rapport":             f"1/{k}",
        "k":                   k,
        "n":                   n,
        **_info_equations(k),
        "reference_n10":       ref_n10,
        "essais_digamma_n10":  essais,
        "cible":               res,
        "premier":             res.get("premier"),
        "premier_indetermine": res.get("premier_indetermine", True),
        "fallback":            res.get("fallback", False),
        "approche":            res.get("approche", "inconnu"),
        "niveau_entier":       not res.get("fallback", False),
        "niveau_geometrique":  res.get("fallback", False),
        "note": (
            f"Système convolutif v7.4 — rapport 1/{k} | "
            f"Approche : {res.get('approche', '?')} | "
            f"n={n} | P={res.get('premier', '?')}"
        ),
        "reference_hol": "methode_spectral.thy § XIII — Pont Savard",
    }


def reconstruire_premier(rapport: Union[str, int],
                          n: int = 10,
                          verifier: bool = True) -> Dict:
    """Reconstruit le premier pour le rapport 1/k à n termes.

    Interface directe — retourne le dict de reconstruct_entier ou
    reconstruct_geometrique selon la disponibilité.

    Args:
        rapport  : '1/k' ou entier k
        n        : nombre de termes (défaut 10 = ancrage de référence)
        verifier : si True, lève AssertionError si (SB-D)/k⁶ ≠ P (niveau entier)
    """
    k = _extraire_k(rapport)
    return reconstruire_avec_fallback_geo(k, n)


def reconstruire_premier_pour_n(rapport: Union[str, int],
                                  n: int) -> Dict:
    """Reconstruit le premier pour une quantité de termes n quelconque.

    Alias direct de reconstruire_avec_fallback_geo — n peut dépasser 10.
    """
    k = _extraire_k(rapport)
    return reconstruire_avec_fallback_geo(k, n)

def verifier_exemples() -> bool:
    """Vérifie les exemples canoniques de l'Excel v7.4.

    Exemples de référence (onglet "Tests Digamma" + "Systeme General") :
      k=3 n=10  → P=227   rang=49
      k=8 n=34  → P=32537 rang=3492  somme_A=5705842489643358455620763423304
      k=4 n=10  → P=947   rang=161
    """
    ok = True

    # Test 1 : ancrage k=3
    r3 = reconstruct_entier(3, 10)
    assert r3["premier"] == 227 and r3["rang_premier"] == 49, \
        f"k=3 n=10 : attendu P=227 rang=49, obtenu P={r3['premier']} rang={r3['rang_premier']}"

    # Test 2 : k=8 n=34 (exemple de validation Excel)
    r8 = reconstruct_entier(8, 34)
    assert r8["premier"] == 32537, \
        f"k=8 n=34 : attendu P=32537, obtenu {r8['premier']}"
    assert r8["rang_premier"] == 3492, \
        f"k=8 n=34 : attendu rang=3492, obtenu {r8['rang_premier']}"
    assert r8["somme_A"] == 5_705_842_489_643_358_455_620_763_423_304, \
        f"k=8 n=34 : somme_A inattendue"
    assert r8["somme_B"] == 45_646_739_917_146_867_644_966_107_124_296, \
        f"k=8 n=34 : somme_B inattendue"
    assert r8["digamma"] == 45_646_739_917_146_867_644_957_577_744_968, \
        f"k=8 n=34 : digamma inattendu"

    # Test 3 : ancrage k=4
    r4 = reconstruct_entier(4, 10)
    assert r4["premier"] == 947 and r4["rang_premier"] == 161, \
        f"k=4 n=10 : attendu P=947 rang=161, obtenu P={r4['premier']} rang={r4['rang_premier']}"

    # Test 4 : essais Digamma k=3 → [227]
    essais3 = digamma_trials_n10(3)
    premiers3 = [t["P"] for t in essais3 if t["premier"]]
    assert premiers3 == [227], \
        f"Essais Digamma k=3 : attendu [227], obtenu {premiers3}"

    # Test 5 : niveau géométrique (facteur g > 0)
    g3 = facteur_g(3)
    assert g3 > 1.0, f"g(3) = {g3} devrait être > 1"
    tA3 = suite_geo_A(3)
    tB3 = suite_geo_B(3)
    assert len(tA3) == 10 and len(tB3) == 10, "Longueur suites géo incorrecte"

    logger.info("verifier_exemples() : TOUS LES TESTS RÉUSSIS ✓")
    return ok
